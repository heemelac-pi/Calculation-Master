    function sanitizeUrl(rawUrl) {
      if (!rawUrl) return '';
      let url = String(rawUrl).trim();
      if (!url) return '';
      
      if (url.toUpperCase().includes('HYPERLINK(')) {
        const m = url.match(/"(https?:\/\/[^"]+)"/i) || url.match(/"([^"]+)"/);
        if (m && m[1]) url = m[1].trim();
      }

      const lower = url.toLowerCase();
      if (lower === 'not found' || lower === 'none' || lower === 'n/a' || lower === '-' || lower === 'null') {
        return '';
      }

      if (!url.includes('.') && !url.includes('/')) {
        return '';
      }

      if (!/^https?:\/\//i.test(url)) {
        url = 'https://' + url;
      }

      if (url.includes('drive.google.com/file/d/')) {
        url = url.replace(/\/view(\?.*)?$/i, '/view?usp=sharing');
      }

      return url;
    }

      const matchedCats = new Set();
      for (let i = 0; i < allData.length; i++) {
        const d = allData[i];
        if (selectedProjects.includes(d.project)) {
          if (d.incCat) matchedCats.add(d.incCat);
          if (d.expCat) matchedCats.add(d.expCat);
          if (d.invType) matchedCats.add(d.invType);
        }
      }
      return fullList.filter(c => matchedCats.has(c));
    }

    function formatPrettyDate(rawDateStr) {
      if (!rawDateStr || !rawDateStr.includes('-')) return rawDateStr;
      const parts = rawDateStr.split('-');
      if (parts.length === 3) {
        const day = parseInt(parts[2], 10);
        const monthIndex = parseInt(parts[1], 10) - 1;
        const year = parts[0];
        return `${day} ${FULL_MONTH_NAMES[monthIndex] || parts[1]} ${year}`;
      }
      return rawDateStr;
    }

    function updateDashDynamicDropdowns() {
      const elYear = document.getElementById('dash-year-select');
      const elFromM = document.getElementById('dash-from-month');
      const elToM = document.getElementById('dash-to-month');
      const durationTextEl = document.getElementById('dash-duration-text');

      const prevYear = elYear.value;
      const prevFromM = elFromM.value;
      const prevToM = elToM.value;

      const uniqueYears = new Set();
      const uniqueMonths = new Set();

      let minDate = "";
      let maxDate = "";

      const isAllProjects = (selectedDashProjects.length === 0 || selectedDashProjects.length === allProjectList.length);

      for (let i = 0; i < allData.length; i++) {
        const d = allData[i];
        if (isAllProjects || selectedDashProjects.includes(d.project)) {
          if (d.date && d.date.includes('/')) {
            const parts = d.date.split('/');
            if (parts[1]) uniqueMonths.add(parts[1]);
            if (parts[2]) uniqueYears.add(parts[2]);
          }
          if (d.rawDate && d.rawDate.includes('-')) {
            if (!minDate || d.rawDate < minDate) minDate = d.rawDate;
            if (!maxDate || d.rawDate > maxDate) maxDate = d.rawDate;
          }
        }
      }

      if (minDate && maxDate) {
        durationTextEl.innerText = `Running: ${formatPrettyDate(minDate)} to ${formatPrettyDate(maxDate)}`;
      } else {
        durationTextEl.innerText = "Running: All Time Records";
      }

      let yearOpts = '<option value="">All Years</option>';
      const sortedYears = Array.from(uniqueYears).sort((a, b) => b - a);
      sortedYears.forEach(y => {
        yearOpts += `<option value="${y}">${y}</option>`;
      });
      elYear.innerHTML = yearOpts;
      elYear.value = uniqueYears.has(prevYear) ? prevYear : "";

      let fromMOpts = '<option value="">From</option>';
      let toMOpts = '<option value="">To</option>';
      const sortedMonths = Array.from(uniqueMonths).sort((a, b) => Number(a) - Number(b));

      sortedMonths.forEach(m => {
        const mName = MONTH_NAMES[parseInt(m, 10) - 1] || m;
        fromMOpts += `<option value="${m}">${mName}</option>`;
        toMOpts += `<option value="${m}">${mName}</option>`;
      });
      elFromM.innerHTML = fromMOpts;
      elToM.innerHTML = toMOpts;
      elFromM.value = uniqueMonths.has(prevFromM) ? prevFromM : "";
      elToM.value = uniqueMonths.has(prevToM) ? prevToM : "";

      const dynamicCats = getCategoriesForProjects(selectedDashProjects, allCategoryList);
      selectedDashCategories = selectedDashCategories.filter(c => dynamicCats.includes(c));
      if (selectedDashCategories.length === 0) {
        selectedDashCategories = [...dynamicCats];
        const selectAllCatChk = document.getElementById('dash-select-all-cat');
        if (selectAllCatChk) selectAllCatChk.checked = true;
      }
      
      renderMultiSelectOptionsForList('dash', 'category', dynamicCats, selectedDashCategories);
    }

    function renderMultiSelectOptions(type, target) {
      let listData = [];
      let selectedList = [];

      if (target === 'project') {
        listData = allProjectList;
        selectedList = (type === 'dash') ? selectedDashProjects : selectedTxProjects;
      } else if (target === 'type') {
        listData = ALL_TYPES_LIST;
        selectedList = (type === 'dash') ? selectedDashTypes : selectedTxTypes;
      } else {
        listData = allCategoryList;
        selectedList = (type === 'dash') ? selectedDashCategories : selectedTxCategories;
      }

      renderMultiSelectOptionsForList(type, target, listData, selectedList);
    }
